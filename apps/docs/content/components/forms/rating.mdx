import { Rating } from 'extraction-ui';

import data from '../data.json';
import { cfg, ratingCfg, ratingControlCfg, ratingIndicatorCfg } from './rating';

import { RatingControlledExample, RatingUncontrolledExample } from './rating-example';

export const Component = <Rating defaultValue={'3'} count={5} />;

export const ComponentCompound = (
  <Rating defaultValue={'3'} count={5}>
    <Rating.Control>
      <Rating.Indicator />
    </Rating.Control>
  </Rating>
);

export const ComponentCompoundFull = (
  <Rating defaultValue={'🍋'} className='gap-4' single>
    {['🍋', '🍊', '🍎', '🍓', '🍉'].map((emoji) => (
      <Rating.Control
        key={emoji}
        value={emoji}
        className='grayscale data-[filled="true"]:grayscale-0'
      >
        <Rating.Indicator>{emoji}</Rating.Indicator>
      </Rating.Control>
    ))}
  </Rating>
);

export const ComponentFractional = (
  <Rating name='rating' defaultValue={'2.5'} count={5} fractions={2} />
);

export const Wrapper = ({ children }) => children;

Use the `single` props to keep only one child highlighted.

## Controlled

Use the `value` and `onValueChange` props when you need to control the value externally.

<Usage
  components={{
    usage: <RatingControlledExample />,
    wrapper: Wrapper,
  }}
  serializedCode={`const RatingExample = () => {
    const [value, setValue] = useState('3');
    return <Rating value={value} onValueChange={setValue} count={5}  />;
  };`}
/>

## Uncontrolled

Use the `name` prop to include the value in the form submission.

<Usage
  components={{
    usage: <RatingUncontrolledExample />,
    wrapper: Wrapper,
  }}
  serializedCode={`<form
    onSubmit={(event) => {
      event.preventDefault();
      const formData = new FormData(event.currentTarget);
      console.log(formData.get('rating'));
  }}>
    <Rating name="rating" defaultValue="3" count={5} fractions={2} />
    <br />
    <Button type="submit" className="palette-neutral variant-surface">
      Submit
    </Button>
  </form>`}
/>
